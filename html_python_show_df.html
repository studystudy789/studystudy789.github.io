📊 CSV + Pandas + Plot Demo

packages = [ "pandas", "matplotlib" ] import pandas as pd import matplotlib.pyplot as plt from io import StringIO from pyscript import display csv_data = """country,year,population Japan,2020,125.8 USA,2020,331.0 China,2020,1402.0 """ df = pd.read_csv(StringIO(csv_data)) display(df, target="df_output") fig, ax = plt.subplots() df.plot(ax=ax, x="country", y="population", kind="bar") ax.set_title("Population by Country (2020)") ax.set_xlabel("Country") ax.set_ylabel("Population (M)") display(fig, target="plot_output")
